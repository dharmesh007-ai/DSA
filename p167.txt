class Solution {
    public int[] twoSum(int[] numbers, int target) {
        int csum=0;
        int[] arr={0,0};
        for(int i=0;i<numbers.length;i++){
            for(int j=i+1;j<numbers.length;j++){
                csum=numbers[i]+numbers[j];
                if(csum==target){
                    arr[0]=i+1;
                    arr[1]=j+1;
                    return arr;
                }
            }
        }
        return arr;
    }
}
